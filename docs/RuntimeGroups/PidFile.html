def path(dir: nil)
  require "tmpdir"
  File.join(dir || Dir.tmpdir, "runtime_groups_pids_#{Process.pid}")
end

def with(dir: nil)
  path = path(dir: dir)
  FileUtils.mkdir_p(File.dirname(path))
  File.write(path, "")
  yield path
ensure
  File.delete(path) if File.exist?(path)
end
